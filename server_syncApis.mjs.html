import iseobj from 'wsemi/src/iseobj.mjs'

import clearXSS from 'wsemi/src/clearXSS.mjs'

import b642str from 'wsemi/src/b642str.mjs'
import genIDSeq from 'wsemi/src/genIDSeq.mjs'
import { digestRow } from 'w-serv-orm/src/baseline.mjs'
import { fieldsExternal, fieldsLocal } from '../src/schema/contracts/apis.mjs'

//外部匯入(外部同步): POST /syncAndReplaceTabs 對 apis 表之處理(原則 11, 規格 外部匯入之定義與機制.md; R-IMPORT-01)
//
//語意: 外部來源把它所管理之某一 group 之「完整清單」送入, 本系統依外部自訂之來源鍵 srcKey, 於表鎖內將該 group 之庫內資料同步成與清單一致——
//  批次有、庫無 → 新增(本系統配新 id); 庫有、批次無 → 刪除; 兩邊皆有而外部擁有欄位不同 → 修改(保留 id); 相同 → 不寫。不再 delAll＋insert(消除 ADR-016 之非原子)。
//欄位歸屬(src/schema/contracts/apis.mjs): 外部擁有(fieldsExternal)依批次寫入; 本系統擁有(fieldsLocal: tokens、authType、authConfigJson、testBaseUrl、default*Json、contentType)匯入一律不寫(修 C4b);
//  timeCreate 只於新增時由 ORM 寫(修 A5); isActive、timeCreate、userIdUpdate 等其餘批次所帶之鍵一律忽略(id 例外: 列帶 id 即整批拒收 idNotAllowed)。
//整批驗證、全有或全無: 缺／空／非字串／重複 srcKey、不合規則之來源鍵、列帶 id、欄位型別不合、會動到範圍外之列 → 整批拒收、零寫入(原因入 srLog 不回顯值)。空批次由呼叫端拒收(errRowsInvalid)。
//  對應規格之固定錯誤鍵: srcKeyInvalid ＝ 本套件既有之 errRowInvalid(srLog 之 field='srcKey'、reason='srcKey'|'unicode'|'xss'|'bytes'|'dup'); idNotAllowed ＝ 新增之 idNotAllowed。
//
//不提供呼叫端相容(原則 11、規格 外部匯入之定義與機制.md §9; 使用者 2026-10-23 裁示): srcKey 即刻必填, 不以 id 代替; 外部不得指定內部 id(列帶 id 即整批拒收); 無「兩者皆無之列每次新增」與整批 delAll＋insert。
//資料遷移(一次性, 不是呼叫端相容): 既有外部列無 srcKey 者(舊同步寫入), 以其 id 為有效來源鍵——批次 srcKey 等於其 id 者視為同一列, 於鎖內補上 srcKey(id 不變、本系統擁有欄位保留, 補值不計入 nUpdate);
//  其後只認 srcKey; 對不上者依差異同步處理(批次無者刪除、批次有而庫無者新增, 引用連帶清理)。
//鎖: 與 saveApi／deleteApi 同一把 apis 表鎖(withTabLocks), 讀取、比對、寫入全在鎖內。
//寫入中途失敗: 以讀取時之快照補償還原(只處理已開始之步驟所涉之列: 刪除本批新增者、重插被改或被刪者之原列), 補償後回讀驗證, 失敗(含 ORM 逐筆 ok:0)則記 srLog 並回 saveWriteFail; 可預見之失敗(格式、型別、重複、跨範圍)全在寫入前攔下。
//log: 不記列內容與來源鍵; ORM 失敗字串中之 id[...] 一律遮蔽(遷移前之既有列其 id 可能即外部來源鍵, 可能為個資)。


//own: 取自有屬性值(不讀原型鏈), 無則 undefined。列物件由外部應用送入, 欄位名可為任意字串(R-OBJ-ITER-01 / R-KEY-01)
let own = (o, k) => (Object.prototype.hasOwnProperty.call(o, k) ? o[k] : undefined)


//restoreB64: 'base64:' 前綴自動還原為字串(外部應用傳送含特殊字元之內容, 如 markdown、JSON 範例)
let restoreB64 = (v) => {
    if (isestr(v) && strleft(v, 7) === 'base64:') {
        return b642str(strdelleft(v, 7))
    }
    return v
}


//cExt: 外部擁有欄位之摘要契約("內容是否改變"只比外部擁有欄位; 以多人同時編輯規格之列摘要方式正規化)
let cExt = { pk: 'id', fields: fieldsExternal }


//pickExt: 取一列之外部擁有欄位(庫內列用)
let pickExt = (row) => {
    let o = {}
    for (let f of fieldsExternal) {
        o[f] = own(row, f)
    }
    return o
}


//hasLoneSurrogate: 是否含孤立代理字元(不合法 Unicode: 經 UTF-8 儲存編碼往返會被改寫為 U+FFFD, 使同一批次每次都被視為有改變)
let hasLoneSurrogate = (s) => typeof s === 'string' && !s.isWellFormed()


//toExtValue: 外部擁有欄位之值 → 寫入值(型別驗證 + 正規化); 回 { ok:false } 表型別不合
//  字串型欄位(schema STRING／TEXT): 字串依原值; 有限數字與布林轉其字串形式; 缺／null → ''; 其餘(物件、陣列、函數…)不合——前端以字串處理(toUpperCase 等)，容器值又會被 ORM 深層合併而使重送恆為「有改變」
//  order(INTEGER): 有限數字或字串依原值; 缺／null → ''; 其餘不合
//  寫入前套與 ORM 相同之字元清理(clearXSS)與合法 Unicode 正規化(toWellFormed), 比對才穩定(否則含特殊字元之列每次匯入都被視為有改變)
let toExtValue = (f, v) => {
    if (v === undefined || v === null) {
        return { ok: true, v: '' }
    }
    if (f === 'order') {
        if (typeof v === 'number' && Number.isFinite(v)) {
            return { ok: true, v }
        }
        if (typeof v === 'string') {
            return { ok: true, v: clearXSS(v).toWellFormed() }
        }
        return { ok: false }
    }
    if (typeof v === 'number' && Number.isFinite(v)) {
        return { ok: true, v: String(v) }
    }
    if (typeof v === 'boolean') {
        return { ok: true, v: String(v) }
    }
    if (typeof v === 'string') {
        return { ok: true, v: clearXSS(v).toWellFormed() }
    }
    return { ok: false }
}


//chunks: 分批(各 ORM 之 $in 條件有上限, 如 wsemi cvOrmFind 之 10000; 共用核心 mustExist 亦每批 1000)
let chunks = (arr, n) => {
    let rs = []
    for (let i = 0; i < arr.length; i += n) {
        rs.push(arr.slice(i, i + n))
    }
    return rs
}
let nIn = 1000


//maskOrm: ORM 失敗之原文入 log 前遮蔽主鍵(w-serv-orm 之失敗字串含 id[...]; 遷移前之既有列其 id 可能即外部來源鍵)
let maskOrm = (err) => {
    if (isestr(err)) {
        return err.slice(0, 200).replace(/id\[[^\]]*\]/g, 'id[*]')
    }
    let name = pickErrName(err)
    let code = pickErrCode(err)
    return `(${typeof err})${name ? ` ${name}` : ''}${code ? ` ${code}` : ''}`
}


//hasFailItem: ORM 之逐筆結果是否含失敗(單筆失敗時整批仍 resolve 並以該筆 ok 為 0 回報)
let hasFailItem = (r) => {
    if (isarr(r)) {
        return r.some((v) => v !== null && typeof v === 'object' && v.ok === 0)
    }
    return r !== null && typeof r === 'object' && r.ok === 0
}


function createSyncApis(deps = {}) {

    //deps
    //woItems: 直讀直寫 ORM(讀取為新鮮讀取、補償還原用); procOrm: 寫入走 procOrm 才會自動補 userIdUpdate／timeCreate／timeUpdate; withTabLocks: apis 表鎖(與 procApis 同一實例);
    //srLog: 事件、範圍、計數入 log(不含個資值)
    //未注入 withTabLocks／woItems／procOrm 時以 TypeError reject, 不寫入(fail-closed)
    let { woItems, procOrm, withTabLocks, srLog } = deps

    let tab = 'apis'

    let logw = (method, o) => {
        if (srLog && typeof srLog[method] === 'function') {
            srLog[method](o)
        }
    }


    /**
     * 外部匯入 apis
     *
     * @param {String} userId 輸入操作者 id(app 權杖之使用者), 寫入 userIdUpdate
     * @param {Object} inp 輸入 { group, rows }, group 為非空字串, rows 為非空陣列
     * @param {String} [keyTable='apis'] 輸入資料表名(僅供 log)
     * @returns {Promise<Object>} 回傳 { nAdd, nUpdate, nDel, nSame }; 失敗 reject 固定語系鍵(errGroupInvalid、errRowsInvalid、errRowInvalid、idNotAllowed、getDataError、saveWriteFail)
     */
    let syncApis = async (userId, inp, keyTable = tab) => {

        //group
        let group = get(inp, 'group', '')
        if (!isestr(group) || clearXSS(group) !== group || hasLoneSurrogate(group)) {
            //group 經 ORM 寫入時會被 clearXSS 改寫, 含標記者寫入後無法再以同一 group 選出而破壞冪等, 一併拒收
            logw('error', { event: 'syncAndReplaceTabs', err: 'errGroupInvalid', keyTable })
            return Promise.reject('errGroupInvalid')
        }

        //rows
        let rows = get(inp, 'rows', [])
        if (!isarr(rows) || rows.length === 0) {
            logw('error', { event: 'syncAndReplaceTabs', err: 'errRowsInvalid', keyTable, group })
            return Promise.reject('errRowsInvalid')
        }

        //rejectRow: 逐列檢核失敗, 列索引、欄位名與原因入 srLog 供整合端查因（回呼叫端只給 key, 同其他錯誤契約）
        let rejectRow = (index, field, reason = '') => {
            logw('error', { event: 'syncAndReplaceTabs', err: 'errRowInvalid', keyTable, group, index, field, ...(reason ? { reason } : {}) })
            console.log(`[API]syncAndReplaceTabs/check rows[${index}]: invalid row (field: ${field}${reason ? `, reason: ${reason}` : ''})`)
            return Promise.reject('errRowInvalid')
        }

        //逐列檢核 ①：每列須為物件（非物件列連 group 都無法寫入）
        for (let k = 0; k < rows.length; k++) {
            if (!iseobj(rows[k])) {
                return rejectRow(k, '(row)')
            }
        }

        //rejectId: 列帶 id → 整批拒收（外部不得指定內部 id; 列索引入 srLog, 不回顯 id 值）
        let rejectId = (index) => {
            logw('error', { event: 'syncAndReplaceTabs', err: 'idNotAllowed', keyTable, group, index, field: 'id' })
            console.log(`[API]syncAndReplaceTabs/check rows[${index}]: id not allowed`)
            return Promise.reject('idNotAllowed')
        }

        //逐列正規化與檢核（鎖外、純函數: 不讀寫資料庫）
        //優先序: 任一列有欄位／來源鍵規則之缺陷（errRowInvalid）先於「列帶 id」（idNotAllowed）——後者待整批通過全部來源鍵規則（含位元組上限與同批重複）後才報，idAt 記第一個帶 id 之列
        let items = []
        let idAt = -1
        for (let k = 0; k < rows.length; k++) {
            let raw = rows[k]

            //外部擁有欄位: 'base64:' 還原 → 型別驗證與正規化(字元清理、合法 Unicode) → 必填檢核(對最終寫入值: 清理後為空者亦不合)
            let ext = {}
            for (let f of fieldsExternal) {
                let tv = toExtValue(f, restoreB64(own(raw, f)))
                if (!tv.ok) {
                    return rejectRow(k, f, 'type')
                }
                ext[f] = tv.v
            }
            for (let f of ['name', 'url']) {
                if (!isestr(ext[f])) {
                    return rejectRow(k, f)
                }
            }
            ext.group = group //範圍欄由請求決定, 不採批次所帶之值

            //來源鍵: srcKey 必填（不提供相容: 缺、null、空字串或非字串皆整批拒收, 不以 id 代替）
            let srcKey = restoreB64(own(raw, 'srcKey'))
            if (!isestr(srcKey)) {
                return rejectRow(k, 'srcKey', 'srcKey')
            }

            //外部不得指定內部 id: 列帶 id（不論值為何）→ 整批拒收（見下方 idAt）
            if (idAt < 0 && own(raw, 'id') !== undefined) {
                idAt = k
            }

            //來源鍵之字元規則: 不得為孤立代理字元(儲存編碼往返會改寫); 不得含 HTML 標記(含白名單標籤, 如 <b>)
            if (hasLoneSurrogate(srcKey)) {
                return rejectRow(k, 'srcKey', 'unicode')
            }
            if (/<[A-Za-z/!?]/.test(srcKey)) {
                return rejectRow(k, 'srcKey', 'xss')
            }

            items.push({ index: k, key: srcKey, ext })
        }

        //來源鍵規則: 不含會被 ORM 改寫之標記、不超過 ORM 主鍵長度上限、同批不重複（R-SYNC-1 之 checkRowIds）
        let badKey = checkRowIds(items.map((v) => ({ id: v.key })), { maxIdBytes: maxIdBytesDef })
        if (badKey !== null) {
            return rejectRow(items[badKey.index].index, 'srcKey', badKey.reason)
        }

        //列帶 id → 整批拒收（上列所有檢核通過後才報，使「欄位／來源鍵缺陷」恆優先於 idNotAllowed）
        if (idAt >= 0) {
            return rejectId(idAt)
        }

        //newRow: 新增列 = 外部擁有欄位 + 來源鍵 + 本系統擁有欄位之初始值 + isActive; timeCreate／timeUpdate／userIdUpdate 由 procOrm 寫
        //本系統擁有欄位初始為空字串（同改造前 ltdtmapping 補 ''; 測試分頁視空 contentType 為不補 Content-Type, ADR-031）
        //id: 由本系統先行配號（補償需預知本批新增之 id）
        let newRow = (item) => {
            let o = { id: genIDSeq(), ...item.ext, srcKey: item.key, isActive: 'y' }
            for (let f of fieldsLocal) {
                o[f] = ''
            }
            return o
        }

        //failOrm: ORM 失敗改回語系鍵, 原文遮蔽主鍵後入 srLog（R-ERR-1; 不同於 ormFail: 遷移前之既有列其主鍵可能即外部來源鍵）
        let failOrm = (step, key) => (err) => {
            logw('error', { event: 'syncAndReplaceTabs-orm', err: key, orm: maskOrm(err), keyTable, group, step })
            return Promise.reject(key)
        }

        //restoreAll: 寫入中途失敗之補償——刪除本批新增者與已開始之步驟所涉之原列, 再把原列（讀取時之快照）重插, 使庫內回到匯入前狀態; 完成後回讀驗證
        //（ORM 之單筆失敗以逐筆 ok:0 回報而非 reject, 故逐筆檢查; 回讀確認新增列已不在、原列皆在）
        let restoreAll = async (idsNew, rowsOld) => {
            let ids = [...idsNew]
            for (let r of rowsOld) {
                ids.push(r.id)
            }
            for (let part of chunks(ids, nIn)) {
                let r = await woItems[tab].del(part.map((id) => ({ id })))
                if (hasFailItem(r)) {
                    throw new Error('restore: delete failed')
                }
            }
            for (let part of chunks(rowsOld, nIn)) {
                await woItems[tab].insert(part)
            }
            let present = new Set()
            for (let part of chunks(ids, nIn)) {
                let rs = await woItems[tab].select({ id: { $in: part } })
                for (let r of rs) {
                    present.add(String(r.id))
                }
            }
            for (let id of idsNew) {
                if (present.has(String(id))) {
                    throw new Error('restore: new row remains')
                }
            }
            for (let r of rowsOld) {
                if (!present.has(String(r.id))) {
                    throw new Error('restore: old row missing')
                }
            }
        }

        return await withTabLocks([tab], async () => {

            //讀取: 新鮮讀取庫內該 group 之全部列
            let existing = await woItems[tab].select({ group })
                .catch(failOrm('select', 'getDataError'))

            //有效來源鍵 = srcKey; 無 srcKey 之既有列（舊同步寫入或後台自建）以其 id 為有效來源鍵——一次性資料遷移: 批次 srcKey 等於其 id 者視為同一列並補上 srcKey（非呼叫端相容）
            let kpExist = new Map()
            for (let e of existing) {
                let ek = isestr(own(e, 'srcKey')) ? e.srcKey : String(e.id)
                if (kpExist.has(ek)) {
                    //庫內同範圍之有效來源鍵重複: 無法唯一配對 → 拒收（不靜默刪改）
                    logw('error', { event: 'syncAndReplaceTabs', err: 'errRowInvalid', keyTable, group, reason: 'dbDuplicate' })
                    return Promise.reject('errRowInvalid')
                }
                kpExist.set(ek, e)
            }

            //比對 → 新增、修改、刪除、不變
            let rowsAdd = []
            let rowsUpd = []
            let rowsOldTouched = [] //被改（含補 srcKey）之原列快照
            let nUpdate = 0
            let nSame = 0
            let matched = new Set()
            for (let v of items) {
                let e = kpExist.get(v.key)
                if (e === undefined) {
                    rowsAdd.push(newRow(v))
                    continue
                }
                matched.add(v.key)
                let same = digestRow(pickExt(e), cExt) === digestRow(v.ext, cExt)
                let needKey = !isestr(own(e, 'srcKey'))
                if (same) {
                    nSame++
                    if (needKey) {
                        //補 srcKey（遷移: 既有外部列以其 id 補上）; 不改外部擁有欄位, 不計入 nUpdate
                        rowsUpd.push({ id: e.id, srcKey: v.key })
                        rowsOldTouched.push(e)
                    }
                }
                else {
                    nUpdate++
                    rowsUpd.push({ id: e.id, srcKey: v.key, ...v.ext })
                    rowsOldTouched.push(e)
                }
            }
            let rowsDel = []
            for (let [ek, e] of kpExist) {
                if (!matched.has(ek)) {
                    rowsDel.push(e)
                }
            }

            let r = { nAdd: rowsAdd.length, nUpdate, nDel: rowsDel.length, nSame }

            //混雜列提示(原則 3, 不得靜默): 被刪者中有無 srcKey 之列(升版前即在此 group 之後台自建列、或舊同步寫入而尚未被遷移命中之列)——批次 srcKey 未等於其 id 者即刪(D6; 前端對此類列亦有標記, README Upgrade Notes)
            let nDelNoKey = rowsDel.filter((e) => !isestr(own(e, 'srcKey'))).length
            if (nDelNoKey > 0) {
                logw('warn', { event: 'syncAndReplaceTabs-delNoSrcKey', keyTable, group, n: nDelNoKey })
            }

            //無任何寫入
            if (rowsAdd.length === 0 && rowsUpd.length === 0 && rowsDel.length === 0) {
                logw('info', { event: 'syncAndReplaceTabs-diff', keyTable, group, ...r })
                return r
            }

            //寫入（全在表鎖內）: 新增 → 修改 → 刪除; 任一步失敗即補償還原（只處理已開始之步驟所涉之列: stage 1 新增、2 修改、3 刪除）
            let stage = 0
            try {
                if (rowsAdd.length > 0) {
                    stage = 1
                    await procOrm(userId, tab, 'insert', rowsAdd).catch(failOrm('insert', 'saveWriteFail'))
                }
                if (rowsUpd.length > 0) {
                    stage = 2
                    await procOrm(userId, tab, 'save', rowsUpd, { mustExist: true }).catch(failOrm('save', 'saveWriteFail'))
                }
                if (rowsDel.length > 0) {
                    stage = 3
                    await procOrm(userId, tab, 'del', rowsDel.map((e) => ({ id: e.id })), { mustExist: true }).catch(failOrm('del', 'saveWriteFail'))
                }
            }
            catch (err) {
                try {
                    let rowsOld = []
                    if (stage >= 2) {
                        rowsOld.push(...rowsOldTouched)
                    }
                    if (stage >= 3) {
                        rowsOld.push(...rowsDel)
                    }
                    await restoreAll(rowsAdd.map((v) => v.id), rowsOld)
                }
                catch (err2) {
                    //無法補償: 已寫入之部分留在庫內, 明確回報（不回成功）並記錄供人工處置
                    logw('error', { event: 'syncAndReplaceTabs-restoreFail', err: 'saveWriteFail', keyTable, group, stage, nAdd: rowsAdd.length, nUpdate: rowsUpd.length, nDel: rowsDel.length })
                }
                return Promise.reject(err)
            }

            logw('info', { event: 'syncAndReplaceTabs-diff', keyTable, group, ...r })
            return r
        })
    }

    return syncApis
}
