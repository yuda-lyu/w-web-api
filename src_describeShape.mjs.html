/**
 * 控制字元換 '?': 避免請求端或使用者資料以 CR/LF 偽造 log 行
 * 涵蓋 C0 控制字元(碼位 < 32)、DEL(127)、C1 控制字元(0x80～0x9F, 含 U+0085 NEL)、行／段分隔(U+2028、U+2029)與雙向文字控制(U+202A～U+202E、U+2066～U+2069; 可使檢視器之顯示順序與實際不同)
 *
 * @param {String} s 輸入字串
 * @returns {String} 回傳上列字元已換成 '?' 之字串
 */
function safeCtl(s) {
    return Array.from(s, (c) => {
        let x = c.charCodeAt(0)
        let isCtl = x < 32 || (x >= 127 && x <= 159) || x === 0x2028 || x === 0x2029 || (x >= 0x202a && x <= 0x202e) || (x >= 0x2066 && x <= 0x2069)
        return isCtl ? '?' : c
    }).join('')
}


//console 診斷之形狀摘要上限: 鍵名最多列 MAX_KEYS 個、單一鍵名至多 MAX_KEY_LEN 字(超出者截斷並標示), 避免巨量鍵名灌爆瀏覽器 console
const MAX_KEYS = 20
const MAX_KEY_LEN = 40


/**
 * 形狀摘要: console 診斷只印型別、鍵名或筆數, 絕不印任何值(規格 D2「預設不輸出資料值」; V20; ADR-061)
 * - null → '(null)'; undefined → '(undefined)'; 字串 → '(typeof string)'(空字串另標 empty, 便於區分「空」與「非字串」); 其他非物件 → '(typeof 型別)'
 * - 陣列 → '(array, length=N)'; 物件 → '(keys: a,b,c)'(自有可列舉之字串鍵, 至多 20 個、單鍵至多 40 字, 控制字元換 '?'; 無鍵 '(keys: none)')
 * - 永不拋錯(存取器、Proxy 陷阱之例外回 '(typeof object)')
 * - 注意: 物件之鍵名若本身即資料(如以使用者 id 為鍵之表)亦會被印出; 此類物件請改印 `.length`／筆數, 不要交給本函數
 *
 * @param {*} v 輸入任意值(使用者物件、回應本體、傳入參數等)
 * @returns {String} 回傳不含值之摘要字串
 */
function describeShape(v) {
    try {
        if (v === null) {
            return '(null)'
        }
        if (v === undefined) {
            return '(undefined)'
        }
        if (typeof v === 'string') {
            return v === '' ? '(typeof string, empty)' : '(typeof string)'
        }
        if (typeof v !== 'object') {
            return `(typeof ${typeof v})`
        }
        if (isarr(v)) {
            return `(array, length=${Number.isInteger(v.length) ? v.length : '?'})` //length 只收整數(敵意之 Proxy 陣列可令 length 回任意值, 不印)
        }
        let ks = Object.keys(v)
        if (ks.length === 0) {
            return '(keys: none)'
        }
        let s = ks.slice(0, MAX_KEYS).map((k) => {
            return safeCtl(k.length > MAX_KEY_LEN ? `${k.slice(0, MAX_KEY_LEN)}...` : k)
        }).join(',')
        if (ks.length > MAX_KEYS) {
            s += `,...+${ks.length - MAX_KEYS}`
        }
        return `(keys: ${s})`
    }
    catch (err) {
        return '(typeof object)'
    }
}


/**
 * 錯誤摘要: console 診斷只印錯誤類別與白名單內之機器代碼, 不印訊息文字(訊息可能含資料值、路徑或驅動原文)
 * 經 wsemi getDiagBriefErr(單一實作, 規格 D1), 不自寫; 回 '(Error)' 或 '(Error ENOENT)'; 字串錯誤(含伺服端回傳之語系鍵)回 '(string)'; 永不拋錯
 *
 * @param {*} err 輸入錯誤或任意值
 * @returns {String} 回傳不含訊息文字之摘要字串
 */
function describeErr(err) {
    let b = getDiagBriefErr(err)
    return b.errCode === undefined ? `(${safeCtl(String(b.errType))})` : `(${safeCtl(String(b.errType))} ${safeCtl(String(b.errCode))})`
}


export { safeCtl, describeShape, describeErr }
